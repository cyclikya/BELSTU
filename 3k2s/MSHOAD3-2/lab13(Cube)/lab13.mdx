--4.1. Числовые
SELECT 
   { [Measures].[Мера 1], 
     [Measures].[Мера 2],
     [Measures].[Мера 3] } ON COLUMNS,
   [DimDeliveryType].[Category].Members ON ROWS
FROM [LogisticsDW]

--4.2. Обработки строковых значений;
SELECT 
   [DimClient].[ClientName].Members ON ROWS,
   [DimClient].[City].Members ON COLUMNS
FROM [LogisticsDW]
WHERE [Measures].[Мера 1]

--4.3. Для наборов;
SELECT 
   { [Measures].[Мера 1] } ON COLUMNS,
   TOPCOUNT([DimDeliveryType].[DeliveryName].Members, 2, [Measures].[Мера 1]) ON ROWS
FROM [LogisticsDW]

--4.4. Для элемента измерения;
WITH MEMBER [Measures].[PreviousAmount] AS
   ([DimDate].[FullDate].PrevMember, [Measures].[Мера 1])
SELECT 
   { [Measures].[Мера 1], 
     [Measures].[PreviousAmount] } ON COLUMNS,
   [DimDate].[FullDate].Members ON ROWS
FROM [LogisticsDW]

--4.5. Для иерархий;
SELECT 
   [Measures].[Мера 1] ON COLUMNS,
   [DimDate].[MonthName].Members ON ROWS
FROM [LogisticsDW]
WHERE [DimDate].[Year].&[2025]
